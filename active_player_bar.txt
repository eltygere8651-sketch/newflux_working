   164	const ActivePlayerControlBar = ({
   165	  isPlaying,
   166	  setIsPlaying,
   167	  isMuted,
   168	  setIsMuted,
   169	  volume,
   170	  setVolume,
   171	  isFullscreen,
   172	  toggleFullscreen,
   173	  isBabyLock,
   174	  setIsBabyLock,
   175	  played,
   176	  onSeekMouseDown,
   177	  onSeekChange,
   178	  onSeekMouseUp,
   179	  duration,
   180	  currentTime,
   181	  onNext,
   182	  isOverlay = false,
   183	}: {
   184	  isPlaying: boolean;
   185	  setIsPlaying: (val: boolean) => void;
   186	  isMuted: boolean;
   187	  setIsMuted: (val: boolean) => void;
   188	  volume: number;
   189	  setVolume: (val: number) => void;
   190	  isFullscreen: boolean;
   191	  toggleFullscreen: () => void;
   192	  isBabyLock: boolean;
   193	  setIsBabyLock: (val: boolean) => void;
   194	  played: number;
   195	  onSeekMouseDown: () => void;
   196	  onSeekChange: (val: number) => void;
   197	  onSeekMouseUp: (val: number) => void;
   198	  duration: number;
   199	  currentTime: number;
   200	  onNext?: () => void;
   201	  isOverlay?: boolean;
   202	}) => {
   203	  const formatTime = (seconds: number) => {
   204	    if (isNaN(seconds) || seconds < 0) return "0:00";
   205	    const date = new Date(seconds * 1000);
   206	    const hh = date.getUTCHours();
   207	    const mm = date.getUTCMinutes();
   208	    const ss = date.getUTCSeconds().toString().padStart(2, "0");
   209	    if (hh > 0) return `${hh}:${mm.toString().padStart(2, "0")}:${ss}`;
   210	    return `${mm}:${ss}`;
   211	  };
   212	
   213	  return (
   214	    <div
   215	      onClick={(e) => e.stopPropagation()}
   216	      className={`flex flex-col w-full absolute bottom-0 left-0 right-0 bg-gradient-to-t from-black/95 via-black/80 to-transparent z-[100] transition-all duration-300 pointer-events-auto ${
   217	        isFullscreen 
   218	          ? "pb-[max(env(safe-area-inset-bottom),24px)] pt-24 px-4 sm:px-8" 
   219	          : "pb-2 pt-8 px-2 sm:px-3"
   220	      }`}
   221	    >
   222	      {/* LEVEL 2: Progress Bar & Timers */}
   223	      <div className={`flex flex-col w-full max-w-4xl mx-auto ${isFullscreen ? "mb-6" : "mb-2"}`}>
   224	        {/* Sleek Progress Bar */}
   225	        <div className={`flex items-center w-full group/seek relative cursor-pointer ${isFullscreen ? "mb-2.5" : "mb-1"}`}>
   226	          <div className="relative h-7 flex items-center w-full">
   227	            <input
   228	              type="range"
   229	              min={0}
   230	              max={0.999999}
   231	              step="any"
   232	              value={played}
   233	              onMouseDown={onSeekMouseDown}
   234	              onChange={(e) => onSeekChange(parseFloat(e.target.value))}
   235	              onMouseUp={(e) => onSeekMouseUp(parseFloat((e.target as any).value))}
   236	              onTouchStart={onSeekMouseDown}
   237	              onTouchEnd={(e) => onSeekMouseUp(parseFloat((e.target as any).value))}
   238	              className="absolute inset-0 w-full h-full opacity-0 z-20 cursor-pointer"
   239	              title="Progreso del vídeo"
   240	            />
   241	            {/* Track Background */}
   242	            <div className={`absolute left-0 right-0 bg-white/20 rounded-full pointer-events-none overflow-hidden transition-all ${isFullscreen ? "h-1 sm:h-1.5 group-hover/seek:h-1.5 sm:group-hover/seek:h-2" : "h-1 group-hover/seek:h-1.5"}`}>
   243	              {/* Played Track */}
   244	              <div 
   245	                className="h-full bg-red-600 transition-all pointer-events-none"
   246	                style={{ width: `${played * 100}%` }}
   247	              />
   248	            </div>
   249	            {/* Thumb indicator */}
   250	            <div 
   251	              className={`absolute bg-red-600 rounded-full pointer-events-none shadow-[0_0_12px_rgba(220,38,38,0.9)] opacity-0 group-hover/seek:opacity-100 transition-opacity transform -translate-x-1/2 ${isFullscreen ? "h-3 w-3 sm:h-4 sm:w-4" : "h-2.5 w-2.5"}`}
   252	              style={{ left: `${played * 100}%` }}
   253	            />
   254	          </div>
   255	        </div>
   256	        {/* Timers */}
   257	        <div className={`flex items-center justify-between font-medium font-mono text-white/90 select-none tracking-wide ${isFullscreen ? "text-[13px] sm:text-sm px-1" : "text-[10px] sm:text-[11px]"}`}>
   258	          <span className="drop-shadow-sm">{formatTime(currentTime)}</span>
   259	          <span className="opacity-70 drop-shadow-sm">{formatTime(duration)}</span>
   260	        </div>
   261	      </div>
   262	
   263	      {/* LEVEL 3: Main Controls Row */}
   264	      <div className={`flex items-center justify-between w-full max-w-4xl mx-auto px-1 sm:px-2 ${isFullscreen ? "gap-1 sm:gap-4" : "gap-1"}`}>
   265	        
   266	        {/* Previous (10s Back) */}
   267	        <button
   268	          type="button"
   269	          onClick={(e) => {
   270	            e.stopPropagation();
   271	            let newPlayed = Math.max(0, (currentTime - 10) / (duration || 1));
   272	            onSeekChange(newPlayed);
   273	            onSeekMouseUp(newPlayed);
   274	          }}
   275	          className={`flex shrink-0 items-center justify-center text-white/90 hover:text-white hover:bg-white/10 active:bg-white/20 rounded-full transition-all cursor-pointer backdrop-blur-sm ${
   276	            isFullscreen ? "w-[48px] h-[48px] sm:w-[56px] sm:h-[56px]" : "w-[32px] h-[32px] sm:w-[36px] sm:h-[36px]"
   277	          }`}
   278	          title="Retroceder 10s"
   279	        >
   280	          <SkipBack className={`fill-current ${isFullscreen ? "w-6 h-6 sm:w-7 sm:h-7" : "w-4 h-4 sm:w-5 sm:h-5"}`} />
   281	        </button>
   282	
   283	        {/* Play / Pause */}
   284	        <button
   285	          type="button"
   286	          onClick={(e) => {
   287	            e.stopPropagation();
   288	            setIsPlaying(!isPlaying);
   289	          }}
   290	          className={`flex shrink-0 items-center justify-center text-white hover:text-white hover:bg-white/10 active:bg-white/20 rounded-full transition-all cursor-pointer backdrop-blur-sm shadow-[0_0_15px_rgba(255,255,255,0.1)] border border-white/5 ${
   291	            isFullscreen ? "w-[48px] h-[48px] sm:w-[56px] sm:h-[56px]" : "w-[36px] h-[36px] sm:w-[40px] sm:h-[40px]"
   292	          }`}
   293	          title={isPlaying ? "Pausar" : "Reproducir"}
   294	        >
   295	          {isPlaying ? (
   296	            <Pause className={`fill-current ${isFullscreen ? "w-7 h-7 sm:w-8 sm:h-8" : "w-5 h-5 sm:w-5 sm:h-5"}`} />
   297	          ) : (
   298	            <Play className={`fill-current ml-0.5 ${isFullscreen ? "w-7 h-7 sm:w-8 sm:h-8" : "w-5 h-5 sm:w-5 sm:h-5"}`} />
   299	          )}
   300	        </button>
   301	        
   302	        {/* Next */}
   303	        <button
   304	          type="button"
   305	          onClick={(e) => {
   306	            e.stopPropagation();
   307	            onNext?.();
   308	          }}
   309	          className={`flex shrink-0 items-center justify-center text-white/90 hover:text-white hover:bg-white/10 active:bg-white/20 rounded-full transition-all cursor-pointer backdrop-blur-sm ${
   310	            isFullscreen ? "w-[48px] h-[48px] sm:w-[56px] sm:h-[56px]" : "w-[32px] h-[32px] sm:w-[36px] sm:h-[36px]"
   311	          }`}
   312	          title="Siguiente video"
   313	        >
   314	          <SkipForward className={`fill-current ${isFullscreen ? "w-6 h-6 sm:w-7 sm:h-7" : "w-4 h-4 sm:w-5 sm:h-5"}`} />
   315	        </button>
   316	
   317	        {/* Volume */}
   318	        <button
   319	          type="button"
   320	          onClick={(e) => {
   321	            e.stopPropagation();
   322	            setIsMuted(!isMuted);
   323	          }}
   324	          className={`flex shrink-0 items-center justify-center text-white/90 hover:text-white hover:bg-white/10 active:bg-white/20 rounded-full transition-all cursor-pointer backdrop-blur-sm ${
   325	            isFullscreen ? "w-[48px] h-[48px] sm:w-[56px] sm:h-[56px]" : "w-[32px] h-[32px] sm:w-[36px] sm:h-[36px]"
   326	          }`}
   327	          title="Volumen"
   328	        >
   329	          {isMuted || volume === 0 ? (
   330	            <VolumeX className={`text-red-400 ${isFullscreen ? "w-6 h-6 sm:w-7 sm:h-7" : "w-4 h-4 sm:w-5 sm:h-5"}`} />
   331	          ) : (
   332	            <Volume2 className={isFullscreen ? "w-6 h-6 sm:w-7 sm:h-7" : "w-4 h-4 sm:w-5 sm:h-5"} />
   333	          )}
   334	        </button>
   335	
   336	        {/* Lock */}
   337	        <button
   338	          type="button"
   339	          onClick={(e) => {
   340	            e.stopPropagation();
   341	            setIsBabyLock(true);
   342	          }}
   343	          className={`flex shrink-0 items-center justify-center text-white/90 hover:text-red-400 hover:bg-white/10 active:bg-white/20 rounded-full transition-all cursor-pointer backdrop-blur-sm ${
   344	            isFullscreen ? "w-[48px] h-[48px] sm:w-[56px] sm:h-[56px]" : "w-[32px] h-[32px] sm:w-[36px] sm:h-[36px]"
   345	          }`}
   346	          title="Bloqueo Infantil"
   347	        >
   348	          <Lock className={isFullscreen ? "w-6 h-6 sm:w-7 sm:h-7" : "w-4 h-4 sm:w-5 sm:h-5"} />
   349	        </button>
   350	
   351	        {/* Fullscreen */}
   352	        <button
   353	          type="button"
   354	          onClick={(e) => {
   355	            e.stopPropagation();
   356	            toggleFullscreen();
   357	          }}
   358	          className={`flex shrink-0 items-center justify-center text-white/90 hover:text-white hover:bg-white/10 active:bg-white/20 rounded-full transition-all cursor-pointer backdrop-blur-sm ${
   359	            isFullscreen ? "w-[48px] h-[48px] sm:w-[56px] sm:h-[56px]" : "w-[32px] h-[32px] sm:w-[36px] sm:h-[36px]"
   360	          }`}
   361	          title={isFullscreen ? "Salir de pantalla completa" : "Pantalla completa"}
   362	        >
   363	          {isFullscreen ? <Minimize className="w-6 h-6 sm:w-7 sm:h-7" /> : <Maximize className="w-4 h-4 sm:w-5 sm:h-5" />}
   364	        </button>
   365	
   366	      </div>
   367	    </div>
   368	  );
   369	};
